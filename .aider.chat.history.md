      return;
    }
    
    // Загружаем данные из таблицы wb_stocks
    const { data: wbStocks, error } = await sb
      .from('wb_stocks')
      .select('*');
      
    if (error) throw error;
    
    // Создаем карту для быстрого поиска по артикулу
    const wbStockMap = new Map();
    (wbStocks || []).forEach(item => {
      wbStockMap.set(item.nm_id, item);
    });

    });
